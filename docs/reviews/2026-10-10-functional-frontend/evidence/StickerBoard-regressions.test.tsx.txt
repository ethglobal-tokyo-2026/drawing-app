it("REVIEW old board queue cannot overwrite the next visit's move", async () => {
  vi.useFakeTimers({ toFake: ["setTimeout", "clearTimeout"] });
  onAPhone();
  const a = boardSticker({ placement: placedAt(0.5) });
  let serverSpot = a.placement;
  let landFirst = () => {};
  let calls = 0;
  const saveStickerPlacement = vi.fn<ApiClient["saveStickerPlacement"]>((stickerId, spots) => {
    const land = () => {
      serverSpot = spots.placement ?? serverSpot;
      return {stickerId, placement: serverSpot, largePlacement: null, seenAt: null, arrivedAt: a.arrivedAt};
    };
    if (++calls === 1) return new Promise(resolve => { landFirst = () => resolve(land()); });
    return Promise.resolve(land());
  });
  const api = boardApi([a], {
    stickerBoard: () => Promise.resolve({ owner: TEST_OWNER, boardStickers: [{...a, placement: serverSpot}] }),
    saveStickerPlacement,
  });
  const first = await openBoard(api);
  const pressFirst = selectByKeys(first.host, a.stickerId);
  pressFirst("ArrowLeft");
  act(() => void vi.advanceTimersByTime(STEP_SAVE_IDLE_MS));
  pressFirst("ArrowLeft");
  act(() => void vi.advanceTimersByTime(STEP_SAVE_IDLE_MS));
  expect(saveStickerPlacement).toHaveBeenCalledTimes(1);
  await act(() => vi.dynamicImportSettled());
  first.unmount();
  const second = await openBoard(api);
  selectByKeys(second.host, a.stickerId)("ArrowRight");
  act(() => void vi.advanceTimersByTime(STEP_SAVE_IDLE_MS));
  await act(async () => {});
  const newest = serverSpot;
  expect(saveStickerPlacement).toHaveBeenCalledTimes(2);
  await act(async () => landFirst());
  expect(saveStickerPlacement).toHaveBeenCalledTimes(3);
  expect(serverSpot).toEqual(newest);
});
