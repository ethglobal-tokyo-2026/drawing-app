     4	 * The Smoothing a fresh sheet starts at, from 0 (Raw) to 100 (Smooth): a light touch, as artists keep
     5	 * Clip Studio's Stabilization low.

     7	export const FIRST_SMOOTHING = 35;
     8	/** How long the line trails a slow nib at Smooth, in ms. */
     9	const SLOW_LAG_MS = 170;
    10	/**
    11	 * Smoothing's share of Smooth to this power scales the slow lag and the half-lag speed, so the low
    12	 * end, where artists draw, steps finely.
    13	 */
    14	const SMOOTHING_CURVE = 2;
    15	/**
    16	 * The nib speed at Smooth, in sheet units per second, at which the line trails by half a slow nib's
    17	 * lag, and less the quicker it goes: jitter shows on a slow stroke, and lag on a quick one. It
    18	 * scales down with Smoothing, so a light one gives way to speed soonest, as Clip Studio's speed
    19	 * options apply only at its low Stabilization.
    20	 */
    21	const HALF_LAG_SPEED = 2000;
    22	/** The nib's speed is smoothed at this cutoff, in Hz, before it sets the line's: quick enough to follow a stroke that speeds up. */
    23	const SPEED_CUTOFF_HZ = 4;
    24	/** A nib that has sent no sample for this many ms has stopped, and its line glides the rest of the way to it. */
    25	export const PAUSE_MS = 40;
    26	/** A held nib's line moves at most this many ms a frame, so its glide starts where it stopped. */
    27	const HOLD_STEP_MS = 1000 / 60;
    28	/** A held nib's line is on it once this close, in sheet units. */
    29	const ON_NIB = 0.1;
    30	/** A lifted nib's line reaches it in steps this many ms apart. */
    31	const CATCH_UP_MS = 8;
    32	/** A lifted nib's line is put on it after this many steps, should it not have arrived. */
    33	const MAX_CATCH_UP_STEPS = 1000;
    34	/** The interval between samples, in ms, until two have come: a display frame's. */
    35	const FIRST_INTERVAL_MS = 1000 / 60;

    37	/** The filter at a Smoothing: its cutoff for a still nib, in Hz, and the Hz each unit per second of speed adds. */
    38	export interface FilterTuning {
    39	  minCutoff: number;
    40	  beta: number;
    41	}
    42	
    43	/** The filter's tuning at a Smoothing from 0 (Raw) to 100 (Smooth); null at Raw, which leaves the nib's samples as they are. */
    44	export function filterTuning(smoothing: number): FilterTuning | null {
    45	  const share = clamp01(smoothing / 100) ** SMOOTHING_CURVE;
    46	  if (share === 0) return null;
    47	  const minCutoff = 1000 / (2 * Math.PI * SLOW_LAG_MS * share);
    48	  return { minCutoff, beta: minCutoff / (HALF_LAG_SPEED * share) };
    49	}
    50	
    51	/** A low-pass step's weight on the new value, for a cutoff in Hz over `dt` seconds. */
    52	const weight = (cutoff: number, dt: number) => 1 / (1 + 1 / (2 * Math.PI * cutoff * dt));
    53	
    54	/**
    55	 * Smoothing, as a One Euro filter (Casiez, Roussel and Vogel, 2012) on the nib's samples, run on their
    56	 * timestamps: a low-pass whose cutoff rises with the nib's speed, so a slow nib's jitter goes and a
    57	 * quick nib's line keeps up. One cutoff from the speed in the plane serves both axes, so a stroke
    58	 * smooths alike in every direction. The line starts where the nib lands; at Raw it is the nib.
    59	 */
    60	export class Stabilizer {
    61	  /** Where the line is. */
    62	  private x: number;
    63	  private y: number;
    64	  /** The nib's latest sample. */
    65	  private nibX: number;
    66	  private nibY: number;
    67	  /** The time the line stands at: the latest sample's, or a later frame's while the nib is held. */
    68	  private t: number;
    69	  /** The nib's velocity, smoothed, in sheet units per second. */
    70	  private vx = 0;
    71	  private vy = 0;
    72	  /** The latest interval between samples, in ms, which a sample stamped no later than the last takes. */
    73	  private interval = FIRST_INTERVAL_MS;
    74	  private readonly tuning: FilterTuning | null;
    75	
    76	  /** `smoothing` runs from 0 (Raw) to 100 (Smooth). */
    77	  constructor(x: number, y: number, t: number, smoothing: number) {
    78	    [this.x, this.y, this.nibX, this.nibY, this.t] = [x, y, x, y, t];
    79	    this.tuning = filterTuning(smoothing);
    80	  }
    81	
    82	  /** The time the line stands at, in the samples' ms. */
    83	  get time(): number {
    84	    return this.t;
    85	  }
    86	
    87	  /** The line is on the nib. */
    88	  get settled(): boolean {
    89	    return this.x === this.nibX && this.y === this.nibY;
    90	  }
    91	
    92	  /** Takes the nib's next sample; returns where the line is now. */
    93	  add(x: number, y: number, t: number): [number, number] {
    94	    if (t > this.t) [this.interval, this.t] = [t - this.t, t];
    95	    this.step(x, y, this.interval / 1000);
    96	    return [this.x, this.y];
    97	  }
    98	
    99	  /**
   100	   * A frame at time `t` passed with no sample from a nib that has stopped: the line glides toward it,
   101	   * a frame's time at a time, and lands on it once close. Returns where the line is now.
   102	   */
   103	  hold(t: number): [number, number] {
   104	    if (t <= this.t || this.settled) return [this.x, this.y];
   105	    this.step(this.nibX, this.nibY, Math.min(t - this.t, HOLD_STEP_MS) / 1000);
   106	    this.t = t;
   107	    if (Math.hypot(this.nibX - this.x, this.nibY - this.y) < ON_NIB)
   108	      [this.x, this.y] = [this.nibX, this.nibY];
   109	    return [this.x, this.y];
   110	  }
   111	
   112	  /** The nib lifted at x, y: where the line goes on its way there, and when, ending on it. */
   113	  finish(x: number, y: number): [x: number, y: number, t: number][] {
   114	    const way: [number, number, number][] = [];
   115	    if (x !== this.nibX || y !== this.nibY) way.push([...this.add(x, y, this.t), this.t]);
   116	    for (let steps = 0; !this.settled && steps < MAX_CATCH_UP_STEPS; steps++)
   117	      way.push([...this.hold(this.t + CATCH_UP_MS), this.t]);
   118	    if (!this.settled) {
   119	      [this.x, this.y] = [x, y];
   120	      way.push([x, y, this.t]);
   121	    }
   122	    return way;
   123	  }
   124	
   125	  /** One step of the filter toward x, y over `dt` seconds. */
   126	  private step(x: number, y: number, dt: number): void {
   127	    const { tuning } = this;
   128	    if (tuning) {
   129	      const toward = weight(SPEED_CUTOFF_HZ, dt);
   130	      this.vx += toward * ((x - this.nibX) / dt - this.vx);
   131	      this.vy += toward * ((y - this.nibY) / dt - this.vy);
   132	      const share = weight(tuning.minCutoff + tuning.beta * Math.hypot(this.vx, this.vy), dt);
   133	      this.x += share * (x - this.x);
   134	      this.y += share * (y - this.y);
   135	    } else [this.x, this.y] = [x, y];
   136	    [this.nibX, this.nibY] = [x, y];
   137	  }
   138	}

   649	    if (queue.length === 0) return;

   651	      const nib = { x: queue[i], y: queue[i + 1], t: queue[i + 3] };
   652	      const [x, y] = stabilizer.add(nib.x, nib.y, nib.t);
   653	      builder.add(x, y, queue[i + 2], stabilizer.time, nib);
   654	    }

   660	   * nib silent for `PAUSE_MS` has stopped: its line glides to it, and once there, curves all the way.
   661	   */
   662	  private catchUp(live: LiveStroke, time: number): void {
   663	    const { stabilizer, builder } = live;
   664	    if (!stabilizer.settled && time - live.t >= PAUSE_MS) {
   665	      const [x, y] = stabilizer.hold(time);
   666	      builder.add(x, y, live.pressure, stabilizer.time, null);
   667	    }

   669	  }

   677	    }

   681	  private endStroke(takeBack: boolean): void {
   682	    const live = this.live;
   683	    if (!live) return;
   684	    this.live = null;
   685	    this.cancelFrame?.();
   686	    this.cancelFrame = null;
   687	    // A pen whose pressure moved senses it: its next strokes start at their first sample's width.
   688	    if (live.builder.pressured) this.pressurePen = true;
   689	    if (takeBack) {
   690	      if (live.before) this.layer.discard(live.before.ink);
   691	      timeOurWork(INK_WORK.replay, () => this.history.repaint());
   692	      return;
   693	    }
   694	    timeOurWork(INK_WORK.paint, () => {
   695	      this.feed(live);
   696	      // The way to the lift point is the line catching up, not the pointer moving: it keeps the width.
   697	      for (const [x, y, t] of live.stabilizer.finish(live.x, live.y))
   698	        live.builder.add(x, y, live.pressure, t, null);
   699	      live.builder.settle(live.x, live.y);

   701	    });

   710	      this.layer.discard(before.ink);
   711	    }
   712	    timeOurWork(INK_WORK.snapshot, () => this.history.commit(live.builder.op));
   713	    this.events.onCommit(live.builder.op);
   714	    this.notifyHistory();
   715	  }

   182	      // Off, a pen draws the brush's size, whatever it reports.

   190	        // pressing pen that reads no pressure, as it can while lifting, keeps its width.
   191	        const widths = this.pressedWidths;

     4	 * A span is cut until each piece strays no further than this from the curve, in sheet units: closer
     5	 * than shows, so close samples add no points and sparse ones still curve.

     7	export const CURVE_FLATNESS = 0.25;
     8	/** A span is halved at most this many times over, so a cusp can't cut it endlessly. */
     9	const MAX_HALVINGS = 6;
    10	/** A cubic Bézier strays from its chord at most this share of its inner points' distance from it. */
    11	const BEZIER_STRAY = 0.75;
    12	/** Centripetal: the knots step by the square root of each chord, so the curve never loops or overshoots. */
    13	const KNOT_POWER = 0.5;
    14	
    15	/** A cubic Bézier's four control points, flat: x, y for each. */
    16	type Bezier = [number, number, number, number, number, number, number, number];
    17	
    18	/** How far a point is from the segment between two others. */
    19	function fromSegment(x: number, y: number, ax: number, ay: number, bx: number, by: number) {
    20	  const [dx, dy] = [bx - ax, by - ay];
    21	  const along = dx === 0 && dy === 0 ? 0 : ((x - ax) * dx + (y - ay) * dy) / (dx * dx + dy * dy);
    22	  const s = Math.min(1, Math.max(0, along));
    23	  return Math.hypot(x - ax - dx * s, y - ay - dy * s);
    24	}
    25	
    26	/** How far a Bézier strays from its chord at most, from where its inner points are. */
    27	function strays(b: Bezier): number {
    28	  const off = (x: number, y: number) => fromSegment(x, y, b[0], b[1], b[6], b[7]);
    29	  return BEZIER_STRAY * Math.max(off(b[2], b[3]), off(b[4], b[5]));
    30	}

    32	/** De Casteljau at the middle: the two halves of a Bézier. */
    33	function halve(b: Bezier): [Bezier, Bezier] {
    34	  const mid = (i: number, j: number) => (b[i] + b[j]) / 2;
    35	  const [ax, ay, bx, by, cx, cy] = [
    36	    mid(0, 2),
    37	    mid(1, 3),
    38	    mid(2, 4),
    39	    mid(3, 5),
    40	    mid(4, 6),
    41	    mid(5, 7),
    42	  ];
    43	  const [dx, dy, ex, ey] = [(ax + bx) / 2, (ay + by) / 2, (bx + cx) / 2, (by + cy) / 2];
    44	  const [mx, my] = [(dx + ex) / 2, (dy + ey) / 2];
    45	  return [
    46	    [b[0], b[1], ax, ay, dx, dy, mx, my],
    47	    [mx, my, ex, ey, cx, cy, b[6], b[7]],
    48	  ];
    49	}

    51	/**
    52	 * The line a stroke paints through its points: a centripetal Catmull-Rom curve, which passes through
    53	 * every point however unevenly they're spaced, cut into pieces straight enough to paint as capsules,
    54	 * with widths and times eased along it. A span is cut once the point after it is
    55	 * known, so the painted line ends a point short until the stroke settles.
    56	 */
    57	export class StrokeCurve {
    58	  /** The stroke's points, flat, `STRIDE` numbers each; the curve appends its pieces' ends. */
    59	  private readonly out: number[];
    60	  /** The latest points the curve passes through, flat like `out`: the last one waits for its span. */
    61	  private keys: number[];
    62	
    63	  /** `out` holds the stroke's first point, already painted. */
    64	  constructor(out: number[]) {
    65	    this.out = out;
    66	    this.keys = out.slice(-STRIDE);
    67	  }
    68	
    69	  /** The line has reached the newest point: none waits for its span. */
    70	  get settled(): boolean {
    71	    return this.keys.length === STRIDE;
    72	  }
    73	
    74	  /** The newest point the curve passes through. */
    75	  get x(): number {
    76	    return this.keys[this.keys.length - STRIDE];
    77	  }
    78	
    79	  get y(): number {
    80	    return this.keys[this.keys.length - STRIDE + 1];
    81	  }
    82	
    83	  /** The curve passes through x, y next, at this width and ms; the span before it is cut. */
    84	  add(x: number, y: number, width: number, t: number): void {
    85	    this.keys.push(x, y, width, t);
    86	    const n = this.keys.length / STRIDE;
    87	    if (n < 3) return;
    88	    this.span(n - 3);
    89	    this.keys = this.keys.slice(-3 * STRIDE);
    90	  }
    91	
    92	  /**
    93	   * The line ends at x, y for now, as the nib pauses or lifts: the waiting point moves there, a step
    94	   * it was too near to take, and its span is cut. The next point starts a fresh curve from it.
    95	   */
    96	  settle(x: number, y: number): void {
    97	    const { keys } = this;
    98	    let last = keys.length - STRIDE;
    99	    if (keys[last] !== x || keys[last + 1] !== y) {
   100	      if (last === 0) {
   101	        keys.push(x, y, keys[2], keys[3]);
   102	        last += STRIDE;
   103	      } else [keys[last], keys[last + 1]] = [x, y];
   104	    }
   105	    if (last > 0) this.span(last / STRIDE - 1);
   106	    this.keys = keys.slice(-STRIDE);
   107	  }
   108	
   109	  /** Cuts the span from key `i` to the next, whose neighbors set its tangents. */
   110	  private span(i: number): void {
   111	    const k = this.keys;
   112	    const n = k.length / STRIDE;
   113	    const at = (j: number) => [k[j * STRIDE], k[j * STRIDE + 1]] as const;
   114	    const [x1, y1] = at(i);
   115	    const [x2, y2] = at(i + 1);
   116	    if (x1 === x2 && y1 === y2) return;
   117	    // Past either end of the run, the curve's neighbor is its own point mirrored.
   118	    const [x0, y0] = i > 0 ? at(i - 1) : [2 * x1 - x2, 2 * y1 - y2];
   119	    const [x3, y3] = i + 2 < n ? at(i + 2) : [2 * x2 - x1, 2 * y2 - y1];
   120	    const knot = (ax: number, ay: number, bx: number, by: number) =>
   121	      Math.max(Math.hypot(bx - ax, by - ay) ** KNOT_POWER, 1e-6);
   122	    const [d0, d1, d2] = [knot(x0, y0, x1, y1), knot(x1, y1, x2, y2), knot(x2, y2, x3, y3)];
   123	    // The Barry–Goldman tangents at each end, scaled to the span, as a Bézier's inner points.
   124	    const tangent = (pa: number, pb: number, pc: number, da: number, db: number) =>
   125	      (d1 * ((pb - pa) / da - (pc - pa) / (da + db) + (pc - pb) / db)) / 3;
   126	    const bezier: Bezier = [
   127	      x1,
   128	      y1,
   129	      x1 + tangent(x0, x1, x2, d0, d1),
   130	      y1 + tangent(y0, y1, y2, d0, d1),
   131	      x2 - tangent(x1, x2, x3, d1, d2),
   132	      y2 - tangent(y1, y2, y3, d1, d2),
   133	      x2,
   134	      y2,
   135	    ];
   136	    const [w1, t1, w2, t2] = [
   137	      k[i * STRIDE + 2],
   138	      k[i * STRIDE + 3],
   139	      k[(i + 1) * STRIDE + 2],
   140	      k[(i + 1) * STRIDE + 3],
   141	    ];
   142	    const cut = (b: Bezier, s0: number, s1: number, halvings: number) => {
   143	      if (halvings < MAX_HALVINGS && strays(b) > CURVE_FLATNESS) {
   144	        const [left, right] = halve(b);
   145	        const mid = (s0 + s1) / 2;
   146	        cut(left, s0, mid, halvings + 1);
   147	        cut(right, mid, s1, halvings + 1);
   148	        return;
   149	      }
   150	      this.out.push(b[6], b[7], w1 + (w2 - w1) * s1, Math.round(t1 + (t2 - t1) * s1));
   151	    };
   152	    cut(bezier, 0, 1, 0);
   153	  }
   154	}

     7	/**
     8	 * The hull of two circles: one round-capped segment whose width can change along it. Every subpath
     9	 * winds the same way, so a whole stroke fills as one union with clean edges.
    10	 */
    11	function capsule(
    12	  g: CanvasRenderingContext2D,
    13	  x0: number,
    14	  y0: number,
    15	  r0: number,
    16	  x1: number,
    17	  y1: number,
    18	  r1: number,
    19	): void {
    20	  const dx = x1 - x0;
    21	  const dy = y1 - y0;
    22	  const d = Math.hypot(dx, dy);
    23	  // One circle inside the other: the bigger one is the whole hull.
    24	  if (d <= Math.abs(r1 - r0) + 1e-3) {
    25	    const [x, y, r] = r0 >= r1 ? [x0, y0, r0] : [x1, y1, r1];
    26	    g.moveTo(x + r, y);
    27	    g.arc(x, y, r, 0, TAU);
    28	    return;
    29	  }
    30	  const a = Math.atan2(dy, dx);
    31	  const phi = Math.acos((r0 - r1) / d);
    32	  g.moveTo(x0 + r0 * Math.cos(a + phi), y0 + r0 * Math.sin(a + phi));
    33	  g.arc(x0, y0, r0, a + phi, a - phi + TAU);
    34	  g.arc(x1, y1, r1, a - phi, a + phi);
    35	  g.closePath();
    36	}

    44	export function paintStroke(

    46	  op: StrokeOp,
    47	  from = 0,
    48	  to = op.pts.length / STRIDE,
    49	): void {

    60	    g.arc(pts[0], pts[1], r, 0, TAU);
    61	    i = 1;
    62	  }

    68	      pts[j - STRIDE + 1],

    70	      pts[j],
    71	      pts[j + 1],
