# Burn Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Holding Burn throws away a sticker in progress from the seal sheet, or a sealed sticker its Original Artist holds and nobody else has received from its detail; a burned sticker shows nowhere, and only the operator can restore it.

**Architecture:** `stickers.burned_at` marks a burned sticker. One condition, `unburned`, keeps it out of every API read, and the image server stops serving a drawing that only burned stickers show. `POST /api/stickers/:stickerId/burn` burns in one transaction and purges the CDN as an 18+ mark does. A sticker in progress burns on the device alone, through the drawing screen's existing reset. One control, `HoldToBurnButton`, serves every burn.

**Tech Stack:** Hono, Drizzle on SQLite, vitest, React 19, Playwright, safaridriver.

**Spec:** `docs/superpowers/specs/2026-10-15-burn.md`

**Where:** a worktree on branch `burn`: `git worktree add .claude/worktrees/burn -b burn main`. Merge to main only from the main checkout (AGENTS.md).

**Lanes:** Tasks 2–7 (API) and Tasks 9–10 (the hold, the sticker in progress) run side by side. Task 11 needs Task 5's route. Task 8 runs alongside and gates Task 12. Tasks 13–15 close.

```ts
    .run();

    return found;
  };

    artistId,
    stickerId,
    row,
    burn: (as = artistId) => test.send("POST", `/api/stickers/${stickerId}/burn`, { as }),
    burnRow: () =>
      test.db
        .update(stickers)
        .set({ burnedAt: test.clock.now() })
```

```ts
/** Whether `contentHash` names a veil an 18+ mark made, which stays public for good. */
const isRecordedVeil = (db: Pick<Db, "select">, contentHash: string) =>
  db
    .select({ veiledHash: veiledImages.veiledHash })
    .from(veiledImages)
    .where(eq(veiledImages.veiledHash, contentHash))
    .get() !== undefined;

/**
 * Whether the files that show a drawing are only for the NSFW opt-in: some unburned sticker sealed
 * with its content hash is NSFW, and none that isn't. Anyone can seal a copy of a public PNG, so
 * marking the copy never hides the drawing another sticker shows; nor any veil ever made, which
 * everyone without the opt-in sees, and which a sticker minted while NSFW names on Sui for good.
 */
export function isNsfwDrawing(db: Pick<Db, "select">, contentHash: string): boolean {
  const marks = db
    .select({ nsfw: stickers.nsfw })
    .from(stickers)
    .where(and(eq(stickers.contentHash, contentHash), unburned))
    .all();
  if (marks.length === 0 || !marks.every(({ nsfw }) => nsfw)) return false;
  return !isRecordedVeil(db, contentHash);
}

/** Whether every sticker sealed with a drawing is burned, so its files show nowhere; never a veil's. */
export function isBurnedDrawing(db: Pick<Db, "select">, contentHash: string): boolean {
  const sealed = db
    .select({ burnedAt: stickers.burnedAt })
    .from(stickers)
    .where(eq(stickers.contentHash, contentHash))
    .all();
  if (sealed.length === 0 || sealed.some(({ burnedAt }) => burnedAt === null)) return false;
  return !isRecordedVeil(db, contentHash);
}

/** Whether anyone may load the files that show a drawing, so the CDN may hold them. */
export const isPublicDrawing = (db: Pick<Db, "select">, contentHash: string) =>
  !isBurnedDrawing(db, contentHash) && !isNsfwDrawing(db, contentHash);
```

```ts
import { gifts, stickerPlacements, stickers } from "@drawing-app/db";
import { and, eq, isNotNull } from "drizzle-orm";
import type { AppDeps } from "../deps.ts";
import { logInfo } from "../diagnostics.ts";
import { giftHoldingSticker } from "../gifts/packaging.ts";
import { oneAtATime } from "../sui/oneAtATime.ts";
import { isPublicDrawing, purgeDueDrawing } from "./nsfwDrawing.ts";

/** Why a burn was refused: its status and code, and what failed. */
export type BurnRefusal =
  | { status: 404; error: "sticker_not_found"; detail: string }
  | { status: 403; error: "not_original_artist" | "not_yours"; detail: string }
  | { status: 409; error: "has_transfer_trail" | "gift_in_transit"; detail: string };

type BurnRow =
  { refused: BurnRefusal } | { burnedNow: boolean; purgeDue: boolean; contentHash: string };

/**
 * The burn's checks and its writes, in one transaction: its row, both layouts of its holder's board
 * left for the sticker tray, and the CDN purge it makes due when it takes its drawing out of public
 * view. A sticker its holder already burned answers as burned.
 */
function burnRow({ db, clock }: AppDeps, userId: string, stickerId: string): BurnRow {
  return db.transaction(
    (tx): BurnRow => {
      const row = tx
        .select({
          artistId: stickers.artistId,
          ownerId: stickers.ownerId,
          contentHash: stickers.contentHash,
          burnedAt: stickers.burnedAt,
        })
        .from(stickers)
        .where(eq(stickers.id, stickerId))
        .get();
      const notFound = (): BurnRow => ({
        refused: { status: 404, error: "sticker_not_found", detail: `No sticker ${stickerId}` },
      });
      if (!row) return notFound();
      if (row.burnedAt !== null) {
        return row.ownerId === userId
          ? { burnedNow: false, purgeDue: false, contentHash: row.contentHash }
          : notFound();
      }
      if (row.artistId !== userId) {
        const detail = `Only sticker ${stickerId}'s Original Artist can burn it`;
        return { refused: { status: 403, error: "not_original_artist", detail } };
      }
      if (row.ownerId !== userId) {
        const detail = `Sticker ${stickerId} is held by someone else`;
        return { refused: { status: 403, error: "not_yours", detail } };
      }
      const received = tx
        .select({ id: gifts.id })
        .from(gifts)
        .where(and(eq(gifts.stickerId, stickerId), eq(gifts.status, "received")))
        .get();
      if (received) {
        const detail = `Sticker ${stickerId} has a Transfer Trail: gift ${received.id} was received`;
        return { refused: { status: 409, error: "has_transfer_trail", detail } };
      }
      const open = giftHoldingSticker(tx, stickerId);
      if (open) {
        const detail = `Sticker ${stickerId} is in gift ${open.id}: ${open.status}, escrow ${open.escrowStatus}`;
        return { refused: { status: 409, error: "gift_in_transit", detail } };
      }
      const wasPublic = isPublicDrawing(tx, row.contentHash);
      const now = clock.now();
      tx.update(stickers).set({ burnedAt: now }).where(eq(stickers.id, stickerId)).run();
      // Off both layouts, so a restore brings it back in the sticker tray; a layout that never
      // placed it stays empty.
      const theirs = and(
        eq(stickerPlacements.userId, userId),
        eq(stickerPlacements.stickerId, stickerId),
      );
      tx.update(stickerPlacements)
        .set({ onBoard: false })
        .where(and(theirs, isNotNull(stickerPlacements.onBoard)))
        .run();
      tx.update(stickerPlacements)
        .set({ largeOnBoard: false })
        .where(and(theirs, isNotNull(stickerPlacements.largeOnBoard)))
        .run();
      const purgeDue = wasPublic && !isPublicDrawing(tx, row.contentHash);
      if (purgeDue) {
        tx.update(stickers).set({ cdnPurgeDueAt: now }).where(eq(stickers.id, stickerId)).run();
      }
      return { burnedNow: true, purgeDue, contentHash: row.contentHash };
    },
    { behavior: "immediate" },
  );
}

/**
 * Burns a sealed sticker for its Original Artist while they hold it, it has no Transfer Trail and no
 * gift holds it: it then shows nowhere. Its rows, files and Sui object stay, for a restore. A burn
 * that takes its drawing out of public view purges the CDN's copies, as an 18+ mark does, and the
 * sweep retries one that fails.
 */
export async function burnSticker(
  deps: AppDeps,
  userId: string,
  stickerId: string,
): Promise<{ refused: BurnRefusal } | { burned: true }> {
  // The 18+ mark's key, so a burn never lands while a mark makes its veil.
  const outcome = await oneAtATime(`nsfw-mark:${stickerId}`, async () =>
    burnRow(deps, userId, stickerId),
  );
  if ("refused" in outcome) return outcome;
  if (outcome.burnedNow) logInfo("sticker.burned", { stickerId, userId });
  if (outcome.purgeDue) await purgeDueDrawing(deps, stickerId, outcome.contentHash);
  else if (outcome.burnedNow) {
    const reason = "an unburned sticker that isn't 18+ shows its drawing, so its files stay public";
    logInfo("cdn.purge.skipped", { stickerId, reason });
  }
  return { burned: true };
}
```

```ts
import { stickers, users, type Db } from "@drawing-app/db";
import { and, asc, eq, isNotNull } from "drizzle-orm";
import { alias } from "drizzle-orm/sqlite-core";

/** A burned sticker as the operator's list shows it. */
export interface BurnedSticker {
  number: number;
  artist: string;
  holder: string;
  burnedAt: Date;
}

/** Who someone is, to the operator: their handle, else their LINE name, else their id. */
const nameOf = (user: typeof users.$inferSelect) =>
  user.handle ? `@${user.handle}` : (user.lineDisplayName ?? user.id);

/** Every burned sticker, the earliest burn first. */
export function listBurnedStickers(db: Db): BurnedSticker[] {
  const artist = alias(users, "artist");
  const holder = alias(users, "holder");
  return db
    .select({ number: stickers.number, burnedAt: stickers.burnedAt, artist, holder })
    .from(stickers)
    .innerJoin(artist, eq(artist.id, stickers.artistId))
    .innerJoin(holder, eq(holder.id, stickers.ownerId))
    .where(isNotNull(stickers.burnedAt))
    .orderBy(asc(stickers.burnedAt))
    .all()
    .flatMap(({ number, burnedAt, artist, holder }) =>
      burnedAt ? [{ number, artist: nameOf(artist), holder: nameOf(holder), burnedAt }] : [],
    );
}

/**
 * Restores the burned sticker numbered `number`: its burn took it off its holder's board, so it comes
 * back in their sticker tray. False when no burned sticker has that number.
 */
export function restoreBurnedSticker(db: Db, number: number): boolean {
  const restored = db
    .update(stickers)
    .set({ burnedAt: null })
    .where(and(eq(stickers.number, number), isNotNull(stickers.burnedAt)))
    .returning({ id: stickers.id })
    .get();
  return restored !== undefined;
}
```

```ts
import {
  useEffect,
  useLayoutEffect,
  useRef,
  useState,
  type KeyboardEvent as ReactKeyboardEvent,
  type MouseEvent as ReactMouseEvent,
  type PointerEvent as ReactPointerEvent,
} from "react";
import { SLOP_OUT } from "./press";

/** How long Burn is held before it burns: long enough that no stray tap or press gets there. */
export const HOLD_TO_BURN_MS = 1200;

const holdKey = (key: string) => key === " " || key === "Enter";

/**
 * Handlers for Burn, which burns only once held for HOLD_TO_BURN_MS: by a finger, a mouse, or Space or
 * Enter held down. Letting go early, sliding off, a cancelled pointer or the page losing focus puts the
 * fire out, and a tap or a click never burns. `holding` is true while it's held.
 */
export function useHoldToBurn(onBurn: () => void, disabled: boolean) {
  const latest = useRef(onBurn);
  useLayoutEffect(() => {
    latest.current = onBurn;
  });
  const timer = useRef<number | undefined>(undefined);
  const [holding, setHolding] = useState(false);
  const putOut = () => {
    if (timer.current === undefined) return;
    clearTimeout(timer.current);
    timer.current = undefined;
    setHolding(false);
  };
  const light = () => {
    if (disabled || timer.current !== undefined) return;
    setHolding(true);
    timer.current = window.setTimeout(() => {
      timer.current = undefined;
      setHolding(false);
      latest.current();
    }, HOLD_TO_BURN_MS);
  };
  useEffect(() => {
    const onVisibility = () => {
      if (document.visibilityState === "hidden") putOut();
    };
    window.addEventListener("blur", putOut);
    window.addEventListener("pagehide", putOut);
    document.addEventListener("visibilitychange", onVisibility);
    return () => {
      putOut();
      window.removeEventListener("blur", putOut);
      window.removeEventListener("pagehide", putOut);
      document.removeEventListener("visibilitychange", onVisibility);
    };
  }, []);
  useEffect(() => {
    if (disabled) putOut();
  }, [disabled]);
  return {
    holding,
    handlers: {
      onPointerDown: (e: ReactPointerEvent<HTMLElement>) => {
        if (e.isPrimary && e.button === 0) light();
      },
      // A finger stays on the button it pressed until it lifts, so sliding off is measured here.
      onPointerMove: (e: ReactPointerEvent<HTMLElement>) => {
        if (timer.current === undefined) return;
        const r = e.currentTarget.getBoundingClientRect();
        const off =
          e.clientX < r.left - SLOP_OUT ||
          e.clientX > r.right + SLOP_OUT ||
          e.clientY < r.top - SLOP_OUT ||
          e.clientY > r.bottom + SLOP_OUT;
        if (off) putOut();
      },
      onPointerUp: putOut,
      onPointerLeave: putOut,
      onPointerCancel: putOut,
      onLostPointerCapture: putOut,
      onKeyDown: (e: ReactKeyboardEvent) => {
        if (!holdKey(e.key)) return;
        e.preventDefault();
        if (!e.repeat) light();
      },
      onKeyUp: (e: ReactKeyboardEvent) => {
        if (holdKey(e.key)) putOut();
      },
      onBlur: putOut,
      // A hold is the point: no long-press menu, and a click alone does nothing.
      onContextMenu: (e: ReactMouseEvent) => e.preventDefault(),
      onClick: (e: ReactMouseEvent) => e.preventDefault(),
    },
  };
}
```
